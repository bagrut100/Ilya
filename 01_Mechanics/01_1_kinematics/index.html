ЦВЕТОВАЯ ПАЛИТРА И СТИЛИ (Modern Light UI) # --------------------------------------------------------- COLOR_BG = RGBColor(248, 250, 252) # Slate 50 (мягкий офф-вайт) COLOR_CARD = RGBColor(255, 255, 255) # Чистый белый COLOR_BORDER = RGBColor(226, 232, 240) # Slate 200 (тонкая рамка) COLOR_TITLE = RGBColor(15, 23, 42) # Slate 900 COLOR_TEXT = RGBColor(51, 65, 85) # Slate 700 COLOR_MUTED = RGBColor(100, 116, 139) # Slate 500 COLOR_ACCENT = RGBColor(37, 99, 235) # Sapphire Blue COLOR_ACCENT_BG = RGBColor(239, 246, 255) # Голубая подложка бейджа COLOR_EMERALD = RGBColor(16, 185, 129) # Emerald COLOR_AMBER = RGBColor(217, 119, 6) # Amber FONT_FAMILY = "Segoe UI" # --------------------------------------------------------- # 2. ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ РЕНДЕРИНГА MATPLOTLIB # --------------------------------------------------------- def render_latex(formula_str, fontsize=28, text_color="#1e293b"): """Рендерит LaTeX-формулу на прозрачном фоне с высоким разрешением.""" fig = plt.figure(figsize=(0.1, 0.1)) fig.patch.set_alpha(0.0) ax = fig.add_axes([0, 0, 1, 1]) ax.patch.set_alpha(0.0) ax.axis("off") t = ax.text(0.5, 0.5, f"${formula_str}$", fontsize=fontsize, color=text_color, ha="center", va="center") fig.canvas.draw() bbox = t.get_window_extent(fig.canvas.get_renderer()) bbox_inches = bbox.transformed(fig.dpi_scale_trans.inverted()) fig.set_size_inches(bbox_inches.width + 0.35, bbox_inches.height + 0.25) buf = io.BytesIO() plt.savefig(buf, format="png", dpi=300, transparent=True, bbox_inches="tight", pad_inches=0.08) plt.close(fig) buf.seek(0) return buf def render_vector_diagram(mode="parallelogram"): """Создаёт чистую векторную иллюстрацию для правила параллелограмма или треугольника.""" fig, ax = plt.subplots(figsize=(4.8, 3.2), dpi=300) fig.patch.set_alpha(0.0) ax.patch.set_alpha(0.0) ax.axis("off") if mode == "parallelogram": # Векторы из общего начала (0, 0) ax.quiver(0, 0, 2.5, 0, angles='xy', scale_units='xy', scale=1, color='#2563eb', width=0.016) ax.quiver(0, 0, 1.2, 1.8, angles='xy', scale_units='xy', scale=1, color='#0ea5e9', width=0.016) # Пунктирные стороны параллелограмма ax.plot([2.5, 3.7], [0, 1.8], '--', color='#94a3b8', lw=1.5) ax.plot([1.2, 3.7], [1.8, 1.8], '--', color='#94a3b8', lw=1.5) # Результирующая диагональ ax.quiver(0, 0, 3.7, 1.8, angles='xy', scale_units='xy', scale=1, color='#10b981', width=0.02) ax.text(1.25, -0.28, r"$\vec{F}_1$", fontsize=15, color='#1e293b', ha='center', weight='bold') ax.text(0.45, 1.05, r"$\vec{F}_2$", fontsize=15, color='#1e293b', ha='center', weight='bold') ax.text(2.1, 1.25, r"$\vec{R} = \vec{F}_1 + \vec{F}_2$", fontsize=15, color='#047857', ha='center', weight='bold') ax.set_xlim(-0.5, 4.3) ax.set_ylim(-0.5, 2.4) elif mode == "triangle": # Сложение "голова к хвосту" ax.quiver(0, 0, 2.6, 0, angles='xy', scale_units='xy', scale=1, color='#2563eb', width=0.016) ax.quiver(2.6, 0, 1.1, 1.8, angles='xy', scale_units='xy', scale=1, color='#0ea5e9', width=0.016) # Замыкающий вектор из начала первого в конец второго ax.quiver(0, 0, 3.7, 1.8, angles='xy', scale_units='xy', scale=1, color='#10b981', width=0.02) ax.text(1.3, -0.28, r"$\vec{F}_1$", fontsize=15, color='#1e293b', ha='center', weight='bold') ax.text(3.35, 0.85, r"$\vec{F}_2$", fontsize=15, color='#1e293b', ha='center', weight='bold') ax.text(1.7, 1.15, r"$\vec{R}$", fontsize=16, color='#047857', ha='center', weight='bold') ax.set_xlim(-0.5, 4.3) ax.set_ylim(-0.5, 2.4) buf = io.BytesIO() plt.savefig(buf, format="png", dpi=300, transparent=True, bbox_inches="tight", pad_inches=0.05) plt.close(fig) buf.seek(0) return buf # --------------------------------------------------------- # 3. ПОСТРОЕНИЕ БАЗОВОГО ШАБЛОНА СЛАЙДА (UI КАРКАС) # --------------------------------------------------------- def create_base_slide(prs, step_num, total_steps, category="פיזיקה תיכונית"): slide = prs.slides.add_slide(prs.slide_layouts[6]) # Заливка фона офф-вайт bg = slide.shapes.add_shape(MSO_SHAPE.RECTANGLE, 0, 0, prs.slide_width, prs.slide_height) bg.fill.solid() bg.fill.fore_color.rgb = COLOR_BG bg.line.fill.background() # Шапка: Бейдж категории (слева вверху) badge = slide.shapes.add_shape(MSO_SHAPE.ROUNDED_RECTANGLE, Inches(0.8), Inches(1.1), Inches(2.2), Inches(0.42)) badge.fill.solid() badge.fill.fore_color.rgb = COLOR_ACCENT_BG badge.line.fill.background() p = badge.text_frame.paragraphs[0] p.text = category p.font.name = FONT_FAMILY p.font.size = Pt(12) p.font.bold = True p.font.color.rgb = COLOR_ACCENT p.alignment = PP_ALIGN.CENTER # Шапка: Счётчик кадров (справа вверху) counter_box = slide.shapes.add_textbox(Inches(5.0), Inches(1.1), Inches(1.7), Inches(0.42)) p_cnt = counter_box.text_frame.paragraphs[0] p_cnt.text = f"{step_num:02d} / {total_steps:02d}" p_cnt.font.name = FONT_FAMILY p_cnt.font.size = Pt(13) p_cnt.font.bold = True p_cnt.font.color.rgb = COLOR_MUTED p_cnt.alignment = PP_ALIGN.RIGHT return slide def add_card(slide, top_in, height_in, bg_color=COLOR_CARD, border_color=COLOR_BORDER): card = slide.shapes.add_shape(MSO_SHAPE.ROUNDED_RECTANGLE, Inches(0.8), Inches(top_in), Inches(5.9), Inches(height_in)) card.fill.solid() card.fill.fore_color.rgb = bg_color if border_color: card.line.color.rgb = border_color card.line.width = Pt(1.2) else: card.line.fill.background() return card # --------------------------------------------------------- # 4.
